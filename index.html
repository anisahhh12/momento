    <script>

        const video = document.getElementById('videoElement');

        const canvas = document.getElementById('captureCanvas');

        const countdownEl = document.getElementById('countdown');

        const snapBtn = document.getElementById('snapButton');

        const actionBtns = document.getElementById('actionButtons');


        let currentImageData = null;


        // 1. Initialize Camera

        async function initCamera() {

            try {

                const stream = await navigator.mediaDevices.getUserMedia({ 

                    video: { facingMode: 'user' }, 

                    audio: false 

                });

                video.srcObject = stream;

            } catch (err) {

                console.error("Camera access error:", err);

                alert("Please allow camera permissions to use the photo booth!");

            }

        }

        initCamera();


        // 2. Countdown Timer

        function startCountdown() {

            let count = 3;

            countdownEl.style.display = 'block';

            countdownEl.innerText = count;

            snapBtn.style.display = 'none';


            let timer = setInterval(() => {

                count--;

                if (count > 0) {

                    countdownEl.innerText = count;

                } else {

                    clearInterval(timer);

                    countdownEl.style.display = 'none';

                    takeSnapshot();

                }

            }, 1000);

        }


        // 3. Take Snapshot

        function takeSnapshot() {

            canvas.width = video.videoWidth || 640;

            canvas.height = video.videoHeight || 480;

            const ctx = canvas.getContext('2d');

            ctx.drawImage(video, 0, 0, canvas.width, canvas.height);

            

            currentImageData = canvas.toDataURL('image/png');

            video.style.display = 'none';

            actionBtns.style.display = 'flex';

        }


        // 4. Retake

        function retakePhoto() {

            video.style.display = 'block';

            actionBtns.style.display = 'none';

            snapBtn.style.display = 'block';

            currentImageData = null;

        }


        // 5. Upload to Google Drive via Apps Script Endpoint

        function uploadPhoto() {

            if (!currentImageData) return;


            // ⚠️ PASTE YOUR GOOGLE APPS SCRIPT WEB APP URL BELOW:

            const GOOGLE_SCRIPT_URL = "YOUR_GOOGLE_APPS_SCRIPT_WEB_APP_URL_HERE";


            alert("Uploading your photo to Google Drive... Please wait!");


            fetch(GOOGLE_SCRIPT_URL, {

                method: 'POST',

                body: JSON.stringify({ image: currentImageData })

            })

            .then(response => response.json())

            .then(data => {

                alert("Success! Photo uploaded to the wedding album folder! 💍✨");

                retakePhoto();

            })

            .catch(error => {

                console.error('Upload Error:', error);

                alert("Upload failed, saving directly to your phone instead!");

                let link = document.createElement('a');

                link.download = 'majlis-photo.png';

                link.href = currentImageData;

                link.click();

                retakePhoto();

            });

        }
